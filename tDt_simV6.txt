#!/usr/bin/env python3 """ Optogenetic TdT DNA synthesis simulator, v6 (clean rebuild of v5.1 design). MODEL (well-mixed compartment, exact stochastic simulation, no fast approximation)   Per-strand, per-base propensity (s^-1):       r_i = a(t) * E_bind * kcat_i * s_i / (1 + sum_j s_j) * stall(len) * tf       s_i = [dNTP_i] / Km_i                      (competitive Michaelis-Menten)       E_bind = [TdT] / ([TdT] + Kd_strand)       (strand occupancy by enzyme)   Light activation a(t): two-state, da/dt = k_act*L(t)*(1-a) - k_rev*(a - dark_leak)       solved exactly per sub-step (L(t) is constant within a sub-step).   Each cycle: fresh reagent mix for the target base, one light pulse at cycle start,   wash carry-over of leftover dNTP, optional reversible terminator + deblock.   Events inside a sub-step use exact Gillespie (memoryless), so the only   approximation is holding a(t) at its sub-step mean. ASSUMPTIONS TO CALIBRATE (defaults copied from v5.1, NOT literature-validated):   kcat, Km, kd_strand_uM, k_act, k_rev, dark_leak, stall_*, term_kcat_factor. Usage:   python tdt_sim_v6.py --selftest   python tdt_sim_v6.py --target ACGTACGTAC --cells 8 --terminators --out run.json   python tdt_sim_v6.py --target ACGT --set bias_ratio=50 --set dark_leak=0.01 """ from __future__ import annotations import argparse import json import multiprocessing as mp import sys from dataclasses import asdict, dataclass import numpy as np BASES = "ACGT" MOL_PER_UM_FL = 6.02214076e23 * 1e-15 * 1e-6  # molecules per (uM * fL) ~ 602 @dataclass class Params:     # compartment     volume_fl: float = 800.0     n_strands: int = 1000     # enzyme (s^-1, uM), per base in order A,C,G,T     tdt_uM: float = 0.8     kd_strand_uM: float = 0.1     kcat: tuple = (8.0, 8.0, 8.0, 8.0)     km_uM: tuple = (100.0, 100.0, 100.0, 100.0)     # reagents     dntp_uM: float = 300.0     cycle_mode: str = "biased"      # target_only | biased | equimolar     bias_ratio: float = 20.0        # target : each non-target (biased mode)     wash_carryover: float = 1e-3    # fraction of leftover dNTP carried to next cycle     # light / activation     cycle_s: float = 10.0     pulse_on_s: float = 1.0         # one pulse at the start of each cycle     substep_s: float = 0.25         # pulse_on_s and cycle_s should be multiples     k_act: float = 2.0              # s^-1 under light     k_rev: float = 0.5              # s^-1 dark reversion     dark_leak: float = 0.05         # residual active fraction in the dark     # chemistry     terminators: bool = False     term_eff: float = 0.99     deblock_eff: float = 1.0     term_kcat_factor: float = 1.0   # terminators are usually much slower for TdT     # stalling of long strands     stall_len: int = 15     stall_factor: float = 0.2     max_len_factor: float = 3.0     # strand capacity = factor * target length def _round(x, rng):     fl = np.floor(x)     return fl + (rng.random(x.shape) < (x - fl)) def simulate_cell(p: Params, target: str, seed) -> dict:     rng = np.random.default_rng(seed)     L, N = len(target), p.n_strands     max_len = int(L * p.max_len_factor) + 1     seqs = np.full((N, max_len), -1, np.int8)     length = np.zeros(N, np.int64)     blocked = np.zeros(N, bool)     kcat, km = np.asarray(p.kcat, float), np.asarray(p.km_uM, float)     mol_per_uM = p.volume_fl * MOL_PER_UM_FL     e_bind = p.tdt_uM / (p.tdt_uM + p.kd_strand_uM)     tf = p.term_kcat_factor if p.terminators else 1.0     dt = p.substep_s     n_sub = int(round(p.cycle_s / dt))     a = p.dark_leak     carry = np.zeros(4)     supplied = consumed = 0     for k in range(L):         t_idx = BASES.index(target[k])         conc = np.zeros(4)         if p.cycle_mode == "equimolar":             conc[:] = p.dntp_uM         elif p.cycle_mode == "biased":             conc[:] = p.dntp_uM / p.bias_ratio             conc[t_idx] = p.dntp_uM         elif p.cycle_mode == "target_only":             conc[t_idx] = p.dntp_uM         else:             raise ValueError(f"unknown cycle_mode {p.cycle_mode}")         fresh = _round(conc * mol_per_uM, rng)         supplied += int(fresh.sum())         mol = fresh + carry         if p.terminators:             blocked[blocked & (rng.random(N) < p.deblock_eff)] = False         w = np.where(blocked | (length >= max_len), 0.0,                      np.where(length >= p.stall_len, p.stall_factor, 1.0))         for s in range(n_sub):             kl = p.k_act if (s * dt) < p.pulse_on_s - 1e-9 else 0.0             den = kl + p.k_rev             a_inf = (kl + p.k_rev * p.dark_leak) / den             a_new = a_inf + (a - a_inf) * np.exp(-den * dt)             f, a = 0.5 * (a + a_new), a_new             tl = 0.0             while True:                 sat = (mol / mol_per_uM) / km                 rb = f * e_bind * tf * kcat * sat / (1.0 + sat.sum())                 r1, wsum = rb.sum(), w.sum()                 if r1 <= 0 or wsum <= 0:                     break                 tl += rng.exponential(1.0 / (r1 * wsum))                 if tl > dt:                     break                 cs = np.cumsum(w)                 i = int(np.searchsorted(cs, rng.random() * cs[-1], side="right"))                 b = min(int(np.searchsorted(np.cumsum(rb), rng.random() * r1, side="right")), 3)                 seqs[i, length[i]] = b                 length[i] += 1                 mol[b] -= 1                 consumed += 1                 if p.terminators and rng.random() < p.term_eff:                     blocked[i] = True                 w[i] = 0.0 if (blocked[i] or length[i] >= max_len) else (                     p.stall_factor if length[i] >= p.stall_len else 1.0)         assert mol.min() >= 0, "negative dNTP count"         carry = _round(mol * p.wash_carryover, rng)     return dict(seqs=seqs, length=length, supplied=supplied, consumed=consumed) # ---------------------------- analysis ---------------------------------------- def levenshtein(a, b) -> int:     if len(a) < len(b):         a, b = b, a     prev = list(range(len(b) + 1))     for i, x in enumerate(a, 1):         cur = [i]         for j, y in enumerate(b, 1):             cur.append(min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (x != y)))         prev = cur     return prev[-1] def max_run(row) -> int:     best = cur = 1 if len(row) else 0     for i in range(1, len(row)):         cur = cur + 1 if row[i] == row[i - 1] else 1         best = max(best, cur)     return best def analyse(res: dict, target: str) -> dict:     tgt = [BASES.index(c) for c in target]     L = len(tgt)     seqs, length = res["seqs"], res["length"]     N = len(length)     edits = np.empty(N)     runs = np.empty(N)     for n in range(N):         row = seqs[n, :length[n]].tolist()         edits[n] = 0 if row == tgt else levenshtein(row, tgt)         runs[n] = max_run(row)     full = length == L     pos_err = ((seqs[full, :L] != np.array(tgt)).mean(axis=0).tolist()                if full.any() else [float("nan")] * L)     return dict(         exact_match=float((edits == 0).mean()),         full_length=float(full.mean()),         mean_edit_distance=float(edits.mean()),         edit_per_target_base=float(edits.mean() / L),         mean_length=float(length.mean()),         mean_max_homopolymer=float(runs.mean()),         mean_dntp_use_per_strand=float(res["consumed"] / N),         per_position_mismatch_full_length=pos_err,     ) # ---------------------------- ensemble ---------------------------------------- def _worker(args):     pdict, target, seed, fasta_n = args     p = Params(**pdict)     res = simulate_cell(p, target, seed)     out = analyse(res, target)     if fasta_n:         out["_fasta"] = ["".join(BASES[x] for x in res["seqs"][n, :res["length"][n]])                          for n in range(min(fasta_n, p.n_strands))]     return out def bootstrap_ci(x, rng, n=2000):     x = np.asarray(x, float)     means = rng.choice(x, size=(n, len(x)), replace=True).mean(axis=1)     return float(np.percentile(means, 2.5)), float(np.percentile(means, 97.5)) def run_ensemble(p: Params, target: str, cells: int, seed: int, procs: int, fasta_n=0):     seeds = [int(s.generate_state(1)[0]) for s in np.random.SeedSequence(seed).spawn(cells)]     jobs = [(asdict(p), target, s, fasta_n if i == 0 else 0) for i, s in enumerate(seeds)]     if procs > 1 and cells > 1:         with mp.Pool(procs) as pool:             outs = pool.map(_worker, jobs)     else:         outs = [_worker(j) for j in jobs]     rng = np.random.default_rng(seed)     summary = {}     for key in outs[0]:         if key in ("per_position_mismatch_full_length", "_fasta"):             continue         v = [o[key] for o in outs]         lo, hi = bootstrap_ci(v, rng) if cells > 1 else (v[0], v[0])         summary[key] = dict(mean=float(np.mean(v)), ci95=[lo, hi])     summary["per_position_mismatch_full_length"] = np.nanmean(         [o["per_position_mismatch_full_length"] for o in outs], axis=0).tolist()     return summary, outs[0].get("_fasta", []) # ---------------------------- self-tests -------------------------------------- def selftest() -> bool:     ok = True     def check(name, cond, info=""):         nonlocal ok         ok &= bool(cond)         print(f"[{'PASS' if cond else 'FAIL'}] {name} {info}")     # 1. Poisson limit: constant activity, no depletion, no stall, single base.     p = Params(n_strands=500, volume_fl=1e6, cycle_mode="target_only", k_act=0.0,                dark_leak=1.0, stall_len=10**9, max_len_factor=200, wash_carryover=0)     res = simulate_cell(p, "A", seed=1)     sat = p.dntp_uM / p.km_uM[0]     lam = (p.tdt_uM / (p.tdt_uM + p.kd_strand_uM)) * p.kcat[0] * sat / (1 + sat) * p.cycle_s     m, v = res["length"].mean(), res["length"].var()     check("Poisson mean", abs(m - lam) < 5 * np.sqrt(lam / 500), f"(got {m:.1f}, expected {lam:.1f})")     check("Poisson variance ~ mean", abs(v / lam - 1) < 0.25, f"(var/lam={v / lam:.2f})")     # 2. Perfect terminators, target-only: exactly +1 per cycle, zero errors.     p = Params(n_strands=200, volume_fl=1e5, cycle_mode="target_only", terminators=True,                term_eff=1.0, deblock_eff=1.0, wash_carryover=0)     out = analyse(simulate_cell(p, "ACGTTA", seed=2), "ACGTTA")     check("Perfect terminators give exact product", out["exact_match"] == 1.0,           f"(exact={out['exact_match']:.3f})")     # 3. Mass conservation under strong depletion.     p = Params(n_strands=500, volume_fl=0.002, cycle_mode="target_only", wash_carryover=0)     r = simulate_cell(p, "AC", seed=3)     check("dNTP fully limiting and never over-consumed", r["consumed"] <= r["supplied"] and r["consumed"] >= 0.9 * r["supplied"] and r["consumed"] == r["length"].sum(),           f"(consumed={r['consumed']}, supplied={r['supplied']})")     # 4. Reproducibility.     p = Params(n_strands=50, terminators=True, volume_fl=1e4)     a = simulate_cell(p, "ACG", seed=7)["length"]     b = simulate_cell(p, "ACG", seed=7)["length"]     check("Same seed -> identical result", np.array_equal(a, b))     return ok def parse_set(items):     out = {}     for it in items or []:         k, v = it.split("=", 1)         try:             v = json.loads(v)         except json.JSONDecodeError:             pass         out[k] = tuple(v) if isinstance(v, list) else v     return out def main():     ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)     ap.add_argument("--target", default="ACGTACGTAC")     ap.add_argument("--cells", type=int, default=4)     ap.add_argument("--seed", type=int, default=12345)     ap.add_argument("--procs", type=int, default=max(1, mp.cpu_count() -
